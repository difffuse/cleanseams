# Build facts reported by `cleanseams_cli --version --json` (engine, compiler, locked
# dependency revisions); rewritten only when they change.
file(READ "${CS_DEPENDENCY_LOCK}" cs_lock_text)
string(JSON cs_dep_count LENGTH "${cs_lock_text}" dependencies)
math(EXPR cs_dep_last "${cs_dep_count} - 1")
set(cs_dep_json "{")
foreach(cs_i RANGE ${cs_dep_last})
  string(JSON cs_dep_name GET "${cs_lock_text}" dependencies ${cs_i} name)
  string(JSON cs_dep_version GET "${cs_lock_text}" dependencies ${cs_i} version)
  string(JSON cs_dep_commit GET "${cs_lock_text}" dependencies ${cs_i} commit)
  if(cs_i GREATER 0)
    string(APPEND cs_dep_json ",")
  endif()
  string(APPEND cs_dep_json "\"${cs_dep_name}\":{\"version\":\"${cs_dep_version}\",\"commit\":\"${cs_dep_commit}\"}")
endforeach()
string(APPEND cs_dep_json "}")
# Identity of the actual native/binding sources, including uncommitted local edits.
execute_process(COMMAND git rev-parse HEAD WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}"
  OUTPUT_VARIABLE cs_source_revision OUTPUT_STRIP_TRAILING_WHITESPACE ERROR_QUIET)
if(NOT cs_source_revision)
  set(cs_source_revision "unversioned")
endif()
file(GLOB_RECURSE cs_identity_files CONFIGURE_DEPENDS
  "${CMAKE_CURRENT_SOURCE_DIR}/include/*.hpp" "${CMAKE_CURRENT_SOURCE_DIR}/src/*.cpp"
  "${CMAKE_CURRENT_SOURCE_DIR}/src/*.c" "${CMAKE_CURRENT_SOURCE_DIR}/src/*.h"
  "${CMAKE_CURRENT_SOURCE_DIR}/src/*.hpp" "${CMAKE_CURRENT_SOURCE_DIR}/bindings/*.cpp"
  "${CMAKE_CURRENT_SOURCE_DIR}/bindings/*.hpp" "${CMAKE_CURRENT_SOURCE_DIR}/cli/*.cpp"
  "${CMAKE_CURRENT_SOURCE_DIR}/cmake/*.cmake" "${CMAKE_CURRENT_SOURCE_DIR}/cmake/*.in")
list(APPEND cs_identity_files "${CMAKE_CURRENT_SOURCE_DIR}/CMakeLists.txt" "${CS_DEPENDENCY_LOCK}")
list(SORT cs_identity_files)
set(cs_identity "")
foreach(cs_file IN LISTS cs_identity_files)
  file(SHA256 "${cs_file}" cs_hash)
  file(RELATIVE_PATH cs_relative "${CMAKE_CURRENT_SOURCE_DIR}" "${cs_file}")
  string(APPEND cs_identity "${cs_relative}:${cs_hash}\n")
endforeach()
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ${cs_identity_files})
string(SHA256 cs_source_tree_sha256 "${cs_identity}")
file(CONFIGURE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/generated/cs_build_info.hpp" CONTENT
"#pragma once\n#define CS_SOURCE_REVISION \"${cs_source_revision}\"\n#define CS_SOURCE_TREE_SHA256 \"${cs_source_tree_sha256}\"\n#define CS_ENGINE_VERSION \"${PROJECT_VERSION}\"\n#define CS_COMPILER \"${CMAKE_CXX_COMPILER_ID} ${CMAKE_CXX_COMPILER_VERSION}\"\n#define CS_DEPENDENCIES_JSON R\"cs(${cs_dep_json})cs\"\n")
target_include_directories(cleanseams_core PUBLIC

  "$<BUILD_INTERFACE:${CMAKE_CURRENT_BINARY_DIR}/generated>"
  "$<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>")

if(CS_BUILD_BINDINGS)
  find_package(Python 3.11 REQUIRED COMPONENTS Interpreter Development.Module)
  set(PYBIND11_FINDPYTHON ON)
  cs_dependency(pybind11)
  add_subdirectory("${pybind11_SOURCE_DIR}" "${pybind11_BINARY_DIR}" EXCLUDE_FROM_ALL)
  # Let FindPython apply the target SOABI. In pybind11's cross-compiling path,
  # suffix variables guessed inside its subdirectory do not reach this scope.
  pybind11_add_module(cleanseams_native WITH_SOABI bindings/module.cpp bindings/arrays.cpp bindings/mesh_analysis.cpp)
  # A binding owns its core. A shared SONAME/install-name is process-global and
  # can silently resolve another addon's library even under Python namespaces.
  # Keep the public shared core for SDK/CLI consumers; compile the same sources
  # with hidden symbols and no DLL exports for each self-contained binding.
  get_target_property(cs_binding_sources cleanseams_core SOURCES)
  add_library(cleanseams_binding_core STATIC ${cs_binding_sources})
  set_target_properties(cleanseams_binding_core PROPERTIES
    CXX_VISIBILITY_PRESET hidden C_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN YES)
  target_compile_definitions(cleanseams_binding_core PUBLIC CLEANSEAMS_CORE_STATIC_DEFINE)
  target_include_directories(cleanseams_binding_core PUBLIC include "${CMAKE_CURRENT_BINARY_DIR}/generated")
  target_link_libraries(cleanseams_binding_core PRIVATE Eigen3::Eigen superlu Threads::Threads
    PUBLIC nlohmann_json::nlohmann_json)
  cs_project_warnings(cleanseams_binding_core)
  target_link_libraries(cleanseams_native PRIVATE cleanseams_binding_core)
  cs_project_warnings(cleanseams_native)
endif()
